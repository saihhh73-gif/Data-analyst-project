SELECT * FROM e_commerce_project.cleaned_data;

-- 1. Total revenue 
select sum(amount) as total_revenue 
from cleaned_data;

-- 2. What is the overall performance of the business?

select count(distinct order_id) as Total_order,count(distinct cust_id) as Total_customers,
sum(qty) as Total_unit_sold,sum(amount) as Total_revenue,
round(sum(amount)/count(distinct order_id),2) as avg_order_value
from cleaned_data; 

-- 3. How does revenue change over time?

SELECT 
    YEAR(date) AS year_,
    MONTH(date) AS month_,
    SUM(amount) AS monthly_revenue
FROM cleaned_data
GROUP BY year_, month_
ORDER BY year_, month_;

-- 4. Which states generate the highest revenue?

select state ,sum(amount) as Total_Revenue
from cleaned_data
group by state
order by Total_Revenue desc ;

-- 5. Which product categories perform best?

select category ,sum(qty) as Total_unit,sum(amount) as Revenue
from cleaned_data
group by category
order by Total_unit  desc  ;

-- 6. Which customer segments contribute most to revenue?

select age_group,gender,sum(qty) as count ,sum(amount) as sales
from cleaned_data
group by age_group,gender
order by count desc ;

-- 7. Which sales channel performs best?
select channel_, sum(qty) as Quntity , sum(amount) as Revenue
from cleaned_data
group by channel_
order  by revenue desc ;

-- 8. What is the distribution of order statuses?

select status, count(*)
from cleaned_data
group by status ;

-- 9. Who are the top customers and Reguler to buyer ?

select cust_id,  sum(qty) as orders  ,sum(order_id) as Ordes_value
from cleaned_data
group by cust_id 
order by Ordes_value desc
limit 3;

-- 10. Which days generate the most revenue?

select channel_, DAYNAME(date) as day_ , sum(amount) as  Revenue
from cleaned_data
group by day_,channel_ 
order by Revenue desc;

-- 11. What is each category’s contribution to total revenue?
select category,sum(qty),sum(amount) as Sale_by_category
from cleaned_data
group by category
order by Sale_by_category desc;
